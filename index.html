// --- TELEGRAM BOT INTEGRATION --- window.saveTelegramConfigFirebase = function() { const token = document.getElementById('tg-bot-token').value.trim(); const chatId = document.getElementById('tg-chat-id').value.trim(); if (!loggedInUser) return; update(ref(db, `${getUserPath()}/telegram`), { token, chatId }).then(() => { alert('הגדרות טלגרם נשמרו בהצלחה!'); updateTelegramStatusBadge(true); }); }; window.sendTelegramMessageFirebase = function() { const text = document.getElementById('tg-msg-text').value.trim(); const statusEl = document.getElementById('tg-send-status'); const token = document.getElementById('tg-bot-token').value.trim(); const chatId = document.getElementById('tg-chat-id').value.trim(); if (!text) { alert('אנא הזיני טקסט להודעה.'); return; } if (!token || !chatId) { alert('יש להזין ולהציל Token ו-Chat ID בהגדרות הבוט תחילה.'); return; } if (statusEl) statusEl.innerText = 'שולח...'; fetch(`https://api.telegram.org/bot${token}/sendMessage`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ chat_id: chatId, text: text }) }) .then(res => res.json()) .then(data => { if (data.ok) { if (statusEl) { statusEl.innerText = 'ההודעה נשלחה בהצלחה! 🚀'; statusEl.className = 'text-xs text-emerald-600 font-bold'; } document.getElementById('tg-msg-text').value = ''; } else { if (statusEl) { statusEl.innerText = 'שגיאה בשליחה: ' + (data.description || 'פרטים שגויים'); statusEl.className = 'text-xs text-rose-600 font-bold'; } } }) .catch(() => { if (statusEl) { statusEl.innerText = 'שגיאת תקשורת בעת השליחה.'; statusEl.className = 'text-xs text-rose-600 font-bold'; } }); }; function loadTelegramConfig() { if (!localState.telegram) return; const tokenEl = document.getElementById('tg-bot-token'); const chatIdEl = document.getElementById('tg-chat-id'); if (tokenEl && localState.telegram.token) tokenEl.value = localState.telegram.token; if (chatIdEl && localState.telegram.chatId) chatIdEl.value = localState.telegram.chatId; updateTelegramStatusBadge(!!(localState.telegram.token && localState.telegram.chatId)); } function updateTelegramStatusBadge(isConfigured) { const badge = document.getElementById('telegram-status-badge'); if (!badge) return; if (isConfigured) { badge.innerText = '🟢 מחובר ופעיל'; badge.className = 'px-2.5 py-1 rounded text-xs bg-emerald-100 text-emerald-800 font-bold'; } else { badge.innerText = '⚪ הגדרות טרם הוזנו'; badge.className = 'px-2.5 py-1 rounded text-xs bg-slate-100 text-slate-700'; } }
